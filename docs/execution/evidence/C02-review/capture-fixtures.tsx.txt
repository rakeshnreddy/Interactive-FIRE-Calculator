 const states={empty:[],usd:[makeAccount({currency:'USD',latestBalanceCents:10000000})],inr:[makeAccount({currency:'INR',latestBalanceCents:800000000})],mixed:[makeAccount({currency:'USD',latestBalanceCents:10000000}),makeAccount({currency:'INR',latestBalanceCents:800000000}),makeAccount({currency:'USD',category:'liability',latestBalanceCents:500000})],long:[makeAccount({currency:'USD',latestBalanceCents:1234567890100})]};
 for(const [name,accounts]of Object.entries(states)) {

 }

writeFileSync('/tmp/c02-browser-fixtures/'+index+'.html', '<!doctype html><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><link rel="stylesheet" href="/assets/index-B1zU0TSC.css"><title>FinPath synthetic UI review</title><div class="app" data-mode="light"><main>'+html+'</main></div>');return html;

    id: `acc_${Math.random().toString(36).slice(2, 8)}`,

    latestBalanceCents: 50_00_00,
    latestBalanceDate: '2026-09-01',

}

    it('shows Unavailable for aggregated metrics and renders per-currency breakdowns without false sums', () => {
      const accounts: FinancialAccount[] = [
        makeAccount({
          category: 'asset',
          currency: 'USD',
          latestBalanceCents: 50_000_00,
          name: 'US Savings'
        }),
        makeAccount({
          category: 'asset',
          currency: 'INR',
          latestBalanceCents: 80_00_000_00,
          name: 'India Mutual Fund'
        }),
        makeAccount({
          category: 'liability',
          currency: 'USD',
          latestBalanceCents: 5_000_00,
          name: 'US Credit Card'
        })
      ];

      const summary = summarizeAccountList(accounts);
      expect(summary.hasMixedCurrencies).toBe(true);
      expect(summary.primaryCurrency).toBeNull();
      // Crucial requirement: aggregated cents MUST be null to prevent adding 50,000 USD to 80,00,000 INR
      expect(summary.assetsCents).toBeNull();
      expect(summary.liabilitiesCents).toBeNull();
      expect(summary.netWorthCents).toBeNull();

      // Per-currency breakdown exists
      expect(summary.currencies).toEqual(['INR', 'USD']);

      expect(summary.byCurrency.USD.liabilitiesCents).toBe(5_000_00);

      const dashHtml = renderToStaticMarkup(

          accounts={accounts}
          cashflow={mockCashflow}
          calculatorResultMessage=""
          goals={[]}
          insights={[]}

          goalMessage=""

          summary={summary}

      // Per-currency breakdown is rendered in small element
      expect(dashHtml).toContain('currency-breakdown');
      expect(dashHtml).toContain('USD: $45,000');
      expect(dashHtml).toContain('INR: ₹8,000,000');

      expect(dashHtml).not.toContain('$8,045,000');
      expect(dashHtml).not.toContain('₹8,045,000');

    it('formats large multi-million and multi-billion amounts without overflow or exponent notation', () => {
      const hugeAccount = makeAccount({
        category: 'asset',
        currency: 'USD',
        latestBalanceCents: 12_345_678_901_00,
        name: 'Family Office Trust'
      });

      const summary = summarizeAccountList([hugeAccount]);
      expect(summary.netWorthCents).toBe(12_345_678_901_00);

  describe('8. Plan Seed Rejection on Currency Mismatch', () => {
    it('rejects seeding when account currency does not match profile/plan currency', () => {
      const testTimeline: SeedTimeline = {
        currentAge: 35,
        planEndAge: 90,
        retirementAge: 60
      };

      const inrAccount = makeAccount({
        category: 'asset',
        currency: 'INR',
        id: 'acc_inr_1',
        latestBalanceCents: 50_00_000_00,
        latestBalanceDate: '2026-09-01',
        name: 'India Account'
      });

      // Seeding a USD plan with an INR account must be rejected
      const seedResult = previewPlanSeed({
        accounts: [inrAccount],
        goal: null,
        plan: initialPlan,
        portfolioSource: 'accounts',
        profile: {
          birthYear: 1991,
          defaultCurrency: 'USD',
          targetRetirementAge: 60,
          updatedAt: '2026-09-01'
        },
        selectedAccountIds: ['acc_inr_1'],
        timeline: testTimeline,
        todayYear: 2026
      });

      expect(seedResult.ok).toBe(false);
      if (!seedResult.ok) {
        expect(seedResult.errors[0]).toContain('India Account uses INR, not USD.');
        expect(seedResult.errorDetails).toBeDefined();
        expect(seedResult.errorDetails![0]).toMatchObject({
          accountName: 'India Account',
          currency: 'INR',
          expectedCurrency: 'USD',
          reason: PLAN_SEED_ERROR_REASONS.CURRENCY_MISMATCH,
          sourceId: 'acc_inr_1'
        });
      }
    });
  });
